# збірка
./build.sh

# п.2 + п.3 одним запуском
sudo OUT=results/O0 ./measure.sh -e ./matmult_O0 1024 5
sudo OUT=results/O3 ./measure.sh -e ./matmult_O3native 1024 5

# п.1 -- FlameGraph, руками, по кроках
perf script -i results/O0/perf.data > out_O0.perf
head -20 out_O0.perf
~/FlameGraph/stackcollapse-perf.pl out_O0.perf > out_O0.folded
head -5 out_O0.folded
~/FlameGraph/flamegraph.pl --title "matmult -O0" out_O0.folded > flame_O0.svg

perf script -i results/O3/perf.data > out_O3.perf
~/FlameGraph/stackcollapse-perf.pl out_O3.perf > out_O3.folded
~/FlameGraph/flamegraph.pl --title "matmult -O3 -march=native" out_O3.folded > flame_O3.svg

# п.4.3 -- диференціальний графік + абсолютні семпли
~/FlameGraph/difffolded.pl out_O0.folded out_O3.folded \
  | ~/FlameGraph/flamegraph.pl > flame_diff.svg
awk '{s+=$NF} END{print FILENAME, s}' out_O0.folded out_O3.folded

# п.2.3 -- анотація асемблера
perf report -i results/O0/perf.data
perf diff results/O0/perf.data results/O3/perf.data

# п.4.1 -- асемблер
sed -n '/^multiply:/,/^\.LFE/p' asm_O0.s > mul_O0.s
sed -n '/^multiply:/,/^\.LFE/p' asm_O3.s > mul_O3.s
wc -l mul_O0.s mul_O3.s
grep -n -E "mulss|addss" mul_O0.s | head -3
grep -n -A4 "vfmadd" mul_O3.s | head -10

# п.4.2 -- порівняння
diff -y results/O0/time-v.txt   results/O3/time-v.txt
diff -y results/O0/perf-stat.txt results/O3/perf-stat.txt
cat results/O0/energy.txt results/O3/energy.txt
